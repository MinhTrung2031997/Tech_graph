```sql
-- pmy parts_all
DROP TABLE IF EXISTS orders;
CREATE TABLE orders (
  id          BIGINT        NOT NULL PRIMARY KEY,
  seller_id   INT           NOT NULL,
  marketplace VARCHAR(8)    NOT NULL,
  status      VARCHAR(10)   NOT NULL,
  amount      DECIMAL(12,2) NOT NULL,
  created_at  DATETIME      NOT NULL,
  KEY orders_created_at_idx (created_at)
) ENGINE = InnoDB;
SET SESSION cte_max_recursion_depth = 1000;
INSERT INTO orders
WITH RECURSIVE a(n) AS (SELECT 0 UNION ALL SELECT n + 1 FROM a WHERE n < 999),
               b(m) AS (SELECT 0 UNION ALL SELECT m + 1 FROM b WHERE m < 999),
               s(g) AS (SELECT a.n * 1000 + b.m + 1 FROM a CROSS JOIN b)
SELECT g,
       CONV(SUBSTR(MD5(CONCAT('s', g)), 1, 8), 16, 10) % 5000 + 1,
       CASE WHEN CONV(SUBSTR(MD5(CONCAT('m', g)), 1, 8), 16, 10) % 10 < 5 THEN 'shopee'
            WHEN CONV(SUBSTR(MD5(CONCAT('m', g)), 1, 8), 16, 10) % 10 < 8 THEN 'tiktok'
            ELSE 'lazada' END,
       CASE WHEN CONV(SUBSTR(MD5(CONCAT('t', g)), 1, 8), 16, 10) % 20 < 2 THEN 'cancelled'
            WHEN CONV(SUBSTR(MD5(CONCAT('t', g)), 1, 8), 16, 10) % 20 = 2 THEN 'returned'
            ELSE 'delivered' END,
       (g * 7907) % 50000 / 100 + 1,
       TIMESTAMP '2024-01-01 00:00:00' + INTERVAL ((g - 1) * 631584 DIV 10000) SECOND
FROM s ORDER BY g;
ALTER TABLE orders ADD KEY orders_seller_created_idx (seller_id, created_at);
ANALYZE TABLE orders;
```

The router (`router.py`), run from the host with Python 3 and PyMySQL (`python3 -m venv venv && ./venv/bin/pip install pymysql`):

```python
"""Application-level sharding demo: orders split across two MySQL databases by seller_id % 2.
Runs each query pattern many times and prints median wall-clock latency (ms) as seen by the application."""
import statistics, sys, time
from concurrent.futures import ThreadPoolExecutor
import pymysql

def conn(db):
    return pymysql.connect(host="127.0.0.1", port=3307, user="root", password="lab", database=db, autocommit=True)

ALL = conn("parts_all")                         # unsharded baseline, same 1M rows
SHARDS = [conn("parts_s0"), conn("parts_s1")]   # shard n holds sellers with seller_id % 2 = n
POOL = ThreadPoolExecutor(max_workers=2)        # one thread per shard for parallel fan-out
PSHARDS = [conn("parts_s0"), conn("parts_s1")]  # separate connections: a pymysql connection is not thread-safe

def shard_for(seller_id):
    """The routing function: the only place that knows the sharding rule."""
    return SHARDS[seller_id % 2]

def q(c, sql, args=None):
    with c.cursor() as cur:
        cur.execute(sql, args)
        return cur.fetchall()

def fan_out_seq(sql, args=None):
    return [q(c, sql, args) for c in SHARDS]

def fan_out_par(sql, args=None):
    return list(POOL.map(lambda c: q(c, sql, args), PSHARDS))

def bench(name, fn, n=50):
    fn()  # warm up
    ts = []
    for _ in range(n):
        t = time.perf_counter(); fn(); ts.append((time.perf_counter() - t) * 1000)
    print(f"{name:28s} median {statistics.median(ts):8.2f} ms")

DEC = ("2025-12-01", "2026-01-01")
SELLER_GMV = "SELECT count(*), sum(amount) FROM orders WHERE seller_id = %s AND created_at >= %s AND created_at < %s"
MKT = ("SELECT marketplace, count(*), sum(amount) FROM orders WHERE created_at >= %s AND created_at < %s "
       "GROUP BY marketplace")

def merge_mkt(parts):
    """Merge partial aggregates: counts and sums add up across shards."""
    out = {}
    for rows in parts:
        for m, c, s in rows:
            a = out.setdefault(m, [0, 0]); a[0] += c; a[1] += s
    return out

if __name__ == "__main__":
    bench("seller month, unsharded", lambda: q(ALL, SELLER_GMV, (1673, *DEC)), 200)
    bench("seller month, routed", lambda: q(shard_for(1673), SELLER_GMV, (1673, *DEC)), 200)
    bench("order by id, unsharded", lambda: q(ALL, "SELECT * FROM orders WHERE id = %s", (777777,)), 200)
    bench("order by id, both shards", lambda: fan_out_seq("SELECT * FROM orders WHERE id = %s", (777777,)), 200)
    bench("marketplace Dec, unsharded", lambda: q(ALL, MKT, DEC), 30)
    bench("marketplace Dec, sequential", lambda: merge_mkt(fan_out_seq(MKT, DEC)), 30)
    bench("marketplace Dec, parallel", lambda: merge_mkt(fan_out_par(MKT, DEC)), 30)
    a = {r[0]: (r[1], r[2]) for r in q(ALL, MKT, DEC)}
    b = merge_mkt(fan_out_par(MKT, DEC))
    print("merge equals unsharded:", all(a[m][0] == b[m][0] and a[m][1] == b[m][1] for m in a))
```

Lab medians: 0.11–0.19, 0.12–0.17, 0.11–0.18, 0.25–0.34, 25–31, 25–28 and 16–17 ms; the merge matched.

Now the merges that go wrong (`topn.py`, next to `router.py`):

```python
"""Cross-shard top-N and pagination: what a per-shard LIMIT gets right and wrong."""
from router import ALL, q, fan_out_par

DAYS = ("SELECT DATE(created_at) d, sum(amount) g FROM orders WHERE created_at >= '2025-01-01' "
        "GROUP BY d ORDER BY g DESC")
truth = q(ALL, DAYS + " LIMIT 10")

naive = {}                                     # each shard's own top 10, added up: wrong
for rows in fan_out_par(DAYS + " LIMIT 10"):
    for d, g in rows: naive[d] = naive.get(d, 0) + g
naive_top = sorted(naive.items(), key=lambda x: -x[1])[:10]

full = {}                                      # every day from every shard, added up: right
for rows in fan_out_par(DAYS):
    for d, g in rows: full[d] = full.get(d, 0) + g
full_top = sorted(full.items(), key=lambda x: -x[1])[:10]

print("naive: right days", len({d for d, _ in naive_top} & {d for d, _ in truth}), "of 10")
print("naive: right totals", sum(1 for d, g in naive_top if (d, g) in set(map(tuple, truth))))
print("full merge equals truth:", [tuple(r) for r in truth] == [(d, g) for d, g in full_top])

PAGE = "SELECT id, created_at FROM orders ORDER BY created_at DESC, id DESC LIMIT %s OFFSET %s"
t = q(ALL, PAGE, (10, 1000))                   # page 101 on the unsharded table
parts = fan_out_par(PAGE, (1010, 0))           # OFFSET + LIMIT rows from every shard
m = sorted([r for rows in parts for r in rows], key=lambda x: (x[1], x[0]), reverse=True)[1000:1010]
print("page 101 equals truth:", list(t) == m, "rows shipped:", sum(len(r) for r in parts))
```

Lab: 3 of 10 days, 0 totals right; full merge right (730 rows); page 101 right with 2,020 rows shipped. Also try sending `LIMIT 10 OFFSET 1000` to both shards and compare.

`move.py` (next to `router.py`) freezes, copies, flips and deletes one seller, then moves it back:

```python
"""Move one seller between application shards using a directory table: freeze, copy, flip, delete."""
import time
from router import conn, bench
DIR = conn("parts_dir"); S = [conn("parts_s0"), conn("parts_s1")]

def shard_of(seller):
    with DIR.cursor() as c:
        c.execute("SELECT shard, moving FROM seller_shard WHERE seller_id = %s", (seller,)); return c.fetchone()

def move(seller, dst):
    src, _ = shard_of(seller); t0 = time.perf_counter()
    with DIR.cursor() as c:      # 1. freeze writes for this seller (the app checks `moving` before writing)
        c.execute("UPDATE seller_shard SET moving = TRUE WHERE seller_id = %s", (seller,))
    with S[src].cursor() as c:   # 2. copy rows through the application
        c.execute("SELECT * FROM orders WHERE seller_id = %s", (seller,)); rows = c.fetchall()
    with S[dst].cursor() as c:
        c.executemany("INSERT INTO orders VALUES (%s,%s,%s,%s,%s,%s)", rows)
    with DIR.cursor() as c:      # 3. flip the directory entry and unfreeze
        c.execute("UPDATE seller_shard SET shard = %s, moving = FALSE WHERE seller_id = %s", (dst, seller))
    t_flip = time.perf_counter()
    with S[src].cursor() as c:   # 4. delete the old copy
        c.execute("DELETE FROM orders WHERE seller_id = %s", (seller,))
    print(f"seller {seller}: {len(rows)} rows {src}->{dst}, writes frozen {1000 * (t_flip - t0):.1f} ms")

move(1673, 0); move(1673, 1)
bench("route via directory", lambda: shard_of(1673), 500)
```

Lab: 247 rows, frozen 358 ms on the first (cold) run and 34 ms on the second; directory lookup 0.29 ms.
