<Tldr>
Uses the containers and the `parts` database from the [Session 1 lab](/blogs/databases/partitioning-sharding/session-1-why-split-data/lab) (PostgreSQL `idx-pg`, MySQL `idx-mysql`, MongoDB `idx-mongo`), each holding the 5M-row `orders` table. The lab creates:
- **PostgreSQL and MySQL:** `orders_p` (24 monthly partitions), `orders_pid` (partitioned by id), and `sellers`, `order_ids` and `order_items`.
- **MongoDB:** `orders_ts` (time-series) and `orders_2024_01` … `orders_2025_12`.

That's about 1.9 GB more in PostgreSQL (the lab database grew from 0.6 GB to 2.5 GB). Before each step, write down how many partitions or indexes you expect to be touched.
</Tldr>

```sql
-- pg: 24 monthly partitions, same columns as orders, no indexes yet
DROP TABLE IF EXISTS orders_p CASCADE;
CREATE TABLE orders_p (LIKE orders INCLUDING DEFAULTS) PARTITION BY RANGE (created_at);
DO $$
DECLARE m date := '2024-01-01';
BEGIN
  WHILE m < '2026-01-01' LOOP
    EXECUTE format('CREATE TABLE orders_p_%s PARTITION OF orders_p FOR VALUES FROM (%L) TO (%L)',
                   to_char(m, 'YYYY_MM'), m, m + interval '1 month');
    m := m + interval '1 month';
  END LOOP;
END $$;
INSERT INTO orders_p SELECT * FROM orders;                          -- 2.6 s in the lab
CREATE INDEX orders_p_id_idx ON orders_p (id);
CREATE INDEX orders_p_created_at_idx ON orders_p (created_at);
VACUUM ANALYZE orders_p;
```

## Exercise 2: building an index online (Concept 2)

Terminal 1 builds, terminal 2 writes 0.5 s later, terminal 3 watches.

```sql
-- pg, terminal 1
CREATE INDEX CONCURRENTLY orders_p_amount_idx ON orders_p (amount);    -- error
CREATE INDEX orders_p_amount_idx ON orders_p (amount);                 -- blocking build
-- terminal 3, during the build
SELECT c.relname, l.mode FROM pg_locks l JOIN pg_class c ON c.oid = l.relation
JOIN pg_stat_activity a ON a.pid = l.pid
WHERE a.query LIKE 'CREATE INDEX orders_p_amount%' AND l.mode = 'ShareLock';
-- terminal 2, during the build
\timing on
INSERT INTO orders_p VALUES (9000001, 1, 'shopee', 'delivered', 10, '2025-06-15');
SELECT count(*) FROM orders_p WHERE created_at >= '2025-06-01' AND created_at < '2025-06-02';
```

Then the online way:

```sql
-- pg, terminal 1
DROP INDEX orders_p_amount_idx;
CREATE INDEX orders_p_amount_idx ON ONLY orders_p (amount);
SELECT indisvalid FROM pg_index WHERE indexrelid = 'orders_p_amount_idx'::regclass;
CREATE INDEX CONCURRENTLY orders_p_2025_06_amount_idx ON orders_p_2025_06 (amount);  -- terminal 2 inserts meanwhile
ALTER INDEX orders_p_amount_idx ATTACH PARTITION orders_p_2025_06_amount_idx;
EXPLAIN (COSTS OFF) SELECT * FROM orders_p WHERE amount = 77.77
  AND created_at >= '2025-06-01' AND created_at < '2025-07-01';
EXPLAIN (COSTS OFF) SELECT * FROM orders_p WHERE amount = 77.77
  AND created_at >= '2025-05-01' AND created_at < '2025-06-01';
```

```bash
# remaining partitions, one at a time
for t in $(docker exec idx-pg psql -U postgres -d parts -Atc "
  SELECT c.relname FROM pg_inherits i JOIN pg_class c ON c.oid = i.inhrelid
  WHERE i.inhparent = 'orders_p'::regclass AND NOT EXISTS (
    SELECT 1 FROM pg_index x JOIN pg_inherits ii ON ii.inhrelid = x.indexrelid
```

## Exercise 3: unique keys (Concept 3)

```sql
-- pg
DELETE FROM orders_p WHERE id > 5000000;
ALTER TABLE orders_p ADD PRIMARY KEY (id);                   -- error
ALTER TABLE orders_p ADD PRIMARY KEY (id, created_at);
INSERT INTO orders_p VALUES (2500000, 9, 'lazada', 'delivered', 5, '2025-06-15 10:00');
SELECT id, created_at, tableoid::regclass FROM orders_p WHERE id = 2500000;
INSERT INTO orders_p VALUES (2500000, 9, 'lazada', 'delivered', 5, '2025-06-15 10:00');   -- error
DELETE FROM orders_p WHERE id = 2500000 AND created_at = '2025-06-15 10:00';

-- the lookup table
CREATE TABLE order_ids (id bigint PRIMARY KEY);
INSERT INTO order_ids SELECT id FROM orders_p;
CREATE FUNCTION orders_p_claim_id() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO order_ids VALUES (NEW.id);
  RETURN NEW;
END $$;
\timing on
INSERT INTO orders_p SELECT g, 1, 'shopee', 'delivered', 10, '2025-12-01'::timestamptz + (g - 6000000) * interval '20 seconds'
FROM generate_series(6000001, 6100000) g;                     -- without trigger
DELETE FROM orders_p WHERE id > 6000000;
CREATE TRIGGER orders_p_claim_id BEFORE INSERT ON orders_p FOR EACH ROW EXECUTE FUNCTION orders_p_claim_id();
INSERT INTO orders_p SELECT g, 1, 'shopee', 'delivered', 10, '2025-12-01'::timestamptz + (g - 6000000) * interval '20 seconds'
FROM generate_series(6000001, 6100000) g;                     -- with trigger
INSERT INTO orders_p VALUES (2500000, 9, 'lazada', 'delivered', 5, '2025-06-15 10:00');   -- error from the trigger

-- partition by the unique column
CREATE TABLE orders_pid (LIKE orders INCLUDING DEFAULTS, PRIMARY KEY (id)) PARTITION BY RANGE (id);
DO $$ BEGIN FOR k IN 0..23 LOOP
  EXECUTE format('CREATE TABLE orders_pid_%s PARTITION OF orders_pid FOR VALUES FROM (%s) TO (%s)',
                 k, k * 208334 + 1, (k + 1) * 208334 + 1);
END LOOP; END $$;
INSERT INTO orders_pid SELECT * FROM orders;
CREATE INDEX ON orders_pid (created_at);
VACUUM ANALYZE orders_pid;
EXPLAIN (ANALYZE, BUFFERS, COSTS OFF)
SELECT count(*) FROM orders_pid WHERE created_at >= '2025-03-01' AND created_at < '2025-03-02';
SELECT min(id), max(id) FROM orders WHERE created_at >= '2025-03-01' AND created_at < '2025-03-02';
-- add that id range to the query above and count the partitions again
```

```sql
-- my
CREATE TABLE orders_bad (id BIGINT NOT NULL PRIMARY KEY, seller_id INT NOT NULL, created_at DATETIME NOT NULL)
  PARTITION BY RANGE COLUMNS (created_at) (PARTITION p0 VALUES LESS THAN ('2024-02-01'));     -- ERROR 1503
```

Expected (lab): both engines refuse `PRIMARY KEY (id)`; the June copy of order 2500000 is accepted; 100k inserts 290 ms without the trigger, 668 ms with; one-day count on `orders_pid` scans 24 partitions, 1 with the id range.

## Exercise 4: foreign keys (Concept 4)

```sql
-- pg
CREATE TABLE sellers (id integer PRIMARY KEY, name text NOT NULL);
```

## Exercise 5: MongoDB layouts (Concept 5)

```js
// mg: time-series copy and one collection per month
db.orders.aggregate([
  { $project: { _id: 0, order_id: '$_id', seller_id: 1, marketplace: 1, status: 1, amount: 1, created_at: 1 } },
  { $out: { db: 'parts', coll: 'orders_ts',
            timeseries: { timeField: 'created_at', metaField: 'marketplace', granularity: 'minutes' } } }]);
for (let y = 2024; y <= 2025; y++) for (let m = 1; m <= 12; m++) {
  const name = `orders_${y}_${String(m).padStart(2, '0')}`;
  db.orders.aggregate([{ $match: { created_at: { $gte: new Date(Date.UTC(y, m - 1, 1)), $lt: new Date(Date.UTC(y, m, 1)) } } },
                       { $out: name }]);
  db.getCollection(name).createIndex({ created_at: 1 });
}
db.system.buckets.orders_ts.countDocuments();
[db.orders.stats(), db.orders_ts.stats(), db.orders_2025_03.stats()]
  .map(s => [s.ns, (s.storageSize / 1048576).toFixed(1), (s.totalIndexSize / 1048576).toFixed(1)]);

// timings: wrap each in a loop of 50 and divide
const march = { created_at: { $gte: ISODate('2025-03-01'), $lt: ISODate('2025-04-01') } };
const gmv = { $group: { _id: '$marketplace', gmv: { $sum: '$amount' } } };
db.orders.aggregate([{ $match: march }, gmv]);
db.orders_ts.aggregate([{ $match: march }, gmv]);
db.orders_2025_03.aggregate([gmv]);
db.orders_ts.find({ order_id: 2500000 }).explain('executionStats').executionStats.totalDocsExamined;

// uniqueness
db.orders_ts.createIndex({ order_id: 1 }, { unique: true });          // error
db.orders_2025_06.insertOne({ _id: 2500000, created_at: ISODate('2025-06-15T10:00:00Z') });   // accepted
db.orders.insertOne({ _id: 2500000, created_at: ISODate('2025-06-15T10:00:00Z') });           // E11000
db.orders_2025_06.deleteOne({ _id: 2500000 });
```
