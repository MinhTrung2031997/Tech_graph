## Exercise 1: a covering index (Concept 1)

```sql
-- pg
CREATE INDEX IF NOT EXISTS orders_customer_idx ON orders (customer_id);
CREATE INDEX orders_cust_created_incl ON orders (customer_id, created_at) INCLUDE (amount);
VACUUM ANALYZE orders;
SET max_parallel_workers_per_gather = 0;

SET enable_indexonlyscan = off;   -- behave as if the index did not cover the query
EXPLAIN (ANALYZE, BUFFERS) SELECT customer_id, created_at, amount FROM orders WHERE customer_id BETWEEN 1 AND 1000;
-- Bitmap Heap Scan · Buffers ≈ 6,400 · ≈ 4 ms
RESET enable_indexonlyscan;
EXPLAIN (ANALYZE, BUFFERS) SELECT customer_id, created_at, amount FROM orders WHERE customer_id BETWEEN 1 AND 1000;
-- Index Only Scan using orders_cust_created_incl · Heap Fetches: 0 · Buffers ≈ 53 · ≈ 0.7 ms
SELECT pg_size_pretty(pg_relation_size('orders_cust_created_incl'));   -- ≈ 39 MB
```

```sql
-- my: page requests with the @b pattern from Session 1, Exercise 4
CREATE INDEX orders_cust_created_amt ON orders (customer_id, created_at, amount);
-- FORCE INDEX (orders_customer_idx):     Using index condition · ≈ 40,000 page requests
-- FORCE INDEX (orders_cust_created_amt): Using where; Using index · ≈ 1,100 page requests
```

```js
// mg
db.orders.createIndex({ customer_id: 1, created_at: -1, amount: 1 }, { name: 'cust_cov' });
const c = p => db.orders.find({ customer_id: { $gte: 1, $lte: 1000 } }, p).hint('cust_cov').explain('executionStats').executionStats.totalDocsExamined;
c({ customer_id: 1, created_at: 1, amount: 1 });          // 10000 (FETCH: _id is returned)
c({ _id: 0, customer_id: 1, created_at: 1, amount: 1 });  // 0 (PROJECTION_COVERED)
```

## Exercise 2: the visibility map (Concept 2)

```sql
-- pg
CREATE EXTENSION IF NOT EXISTS pg_visibility;
SELECT count(*) FILTER (WHERE all_visible) AS all_visible, count(*) AS pages FROM pg_visibility_map('orders');
-- all pages all-visible after the VACUUM above

UPDATE orders SET status = status WHERE id % 100 = 0;     -- 1% of rows, one per page
SELECT count(*) FILTER (WHERE all_visible) AS all_visible FROM pg_visibility_map('orders');   -- ≈ 100
EXPLAIN (ANALYZE, BUFFERS) SELECT customer_id, created_at, amount FROM orders WHERE customer_id BETWEEN 1 AND 1000;
-- Index Only Scan · Heap Fetches ≈ 9,967 · Buffers ≈ 10,000 · ≈ 34 ms

VACUUM (VERBOSE) orders;
-- "index scan needed: 10000 pages from table (97.99% of total) …"
EXPLAIN (ANALYZE, BUFFERS) SELECT customer_id, created_at, amount FROM orders WHERE customer_id BETWEEN 1 AND 1000;
-- Heap Fetches: 0 again
```

## Exercise 3: partial indexes (Concept 3)

```sql
-- pg
CREATE INDEX orders_status_created_idx ON orders (status, created_at);   -- skip if it exists
CREATE INDEX orders_cancelled_created ON orders (created_at) WHERE status = 'cancelled';
ANALYZE orders;
SELECT relname, pg_size_pretty(pg_relation_size(oid)) FROM pg_class
WHERE relname IN ('orders_status_created_idx', 'orders_cancelled_created');   -- ≈ 35 MB vs 1.1 MB

EXPLAIN (ANALYZE, BUFFERS) SELECT * FROM orders WHERE status = 'cancelled' ORDER BY created_at LIMIT 100;
-- Index Scan using orders_cancelled_created · ≈ 23 buffers

PREPARE q(text) AS SELECT * FROM orders WHERE status = $1 ORDER BY created_at LIMIT 100;
SET plan_cache_mode = force_generic_plan;
EXPLAIN EXECUTE q('cancelled');          -- uses orders_status_created_idx, not the partial index
PREPARE q2(timestamptz) AS SELECT * FROM orders WHERE status = 'cancelled' AND created_at >= $1 ORDER BY created_at LIMIT 100;
EXPLAIN EXECUTE q2('2025-06-01');        -- uses orders_cancelled_created
RESET plan_cache_mode; DEALLOCATE ALL;

CREATE UNIQUE INDEX one_pending_per_customer ON orders (customer_id) WHERE status = 'pending';
-- ERROR: could not create unique index … Key (customer_id)=(…) is duplicated.
```

```sql
-- my: no partial indexes; the CASE workaround indexes every row
CREATE INDEX orders_pending_case ON orders ((CASE WHEN status = 'pending' THEN created_at END));
ANALYZE TABLE orders;
SELECT index_name, stat_value FROM mysql.innodb_index_stats
WHERE table_name = 'orders' AND stat_name = 'size' AND index_name = 'orders_pending_case';   -- ≈ 1,060 pages
DROP INDEX orders_pending_case ON orders;
```

```js
// mg
db.orders.createIndex({ created_at: 1 }, { name: 'pending_created', partialFilterExpression: { status: 'pending' } });
db.orders.stats().indexSizes.pending_created;   // ≈ 2.4 MB
db.orders.find({ created_at: { $gte: ISODate('2025-06-01') } }).sort({ created_at: 1 }).limit(100)
  .explain().queryPlanner.winningPlan.stage;     // 'SORT' over a COLLSCAN: the partial index is not a candidate
```

## Exercise 4: expression indexes (Concept 4)

```sql
-- pg
CREATE INDEX ON orders ((created_at::date));
-- ERROR: functions in index expression must be marked IMMUTABLE
CREATE INDEX orders_day_utc ON orders (((created_at AT TIME ZONE 'UTC')::date));
CREATE INDEX orders_sku_lower ON orders (lower(sku));
ANALYZE orders;
EXPLAIN ANALYZE SELECT * FROM orders WHERE (created_at AT TIME ZONE 'UTC')::date = '2025-03-01';   -- Index Scan, 2,740 rows
EXPLAIN ANALYZE SELECT * FROM orders WHERE lower(sku) = 'sku-000123';                              -- Bitmap on orders_sku_lower
EXPLAIN ANALYZE SELECT * FROM orders WHERE upper(sku) = 'SKU-000123';                              -- not usable: Seq Scan
```

```sql
-- my
CREATE INDEX orders_day ON orders ((DATE(created_at)));
EXPLAIN SELECT * FROM orders WHERE DATE(created_at) = '2025-03-01';   -- ref, orders_day, rows 2740
CREATE INDEX orders_sku_plain ON orders (sku);
SELECT COUNT(*) FROM orders WHERE sku = 'sku-000123';                 -- 200: the default collation is case-insensitive
DROP INDEX orders_sku_plain ON orders;
```

```js
// mg
db.orders.createIndex({ sku: 1 }, { name: 'sku_ci', collation: { locale: 'en', strength: 2 } });
db.orders.find({ sku: 'sku-000123' }).collation({ locale: 'en', strength: 2 }).count();   // 200
db.orders.find({ sku: 'sku-000123' }).count();                                            // 0
```

## Exercise 5: unique indexes, NULLs and upserts (Concept 5)

```sql
-- pg
CREATE TABLE u (id int PRIMARY KEY, email text UNIQUE);
INSERT INTO u VALUES (1, NULL), (2, NULL), (3, 'a@x');   -- OK: NULLs are distinct
INSERT INTO u VALUES (4, 'a@x');                          -- ERROR: duplicate key … (email)=(a@x)
CREATE TABLE u2 (id int PRIMARY KEY, email text UNIQUE NULLS NOT DISTINCT);
INSERT INTO u2 VALUES (1, NULL);
INSERT INTO u2 VALUES (2, NULL);                          -- ERROR: … (email)=(null) already exists
CREATE TABLE up (k int PRIMARY KEY, n int);
INSERT INTO up VALUES (1, 1) ON CONFLICT (k) DO UPDATE SET n = up.n + 1;
INSERT INTO up VALUES (1, 1) ON CONFLICT (k) DO UPDATE SET n = up.n + 1 RETURNING *;   -- 1 | 2
DROP TABLE u, u2, up;
```

```sql
-- my
CREATE TABLE u (id INT PRIMARY KEY, email VARCHAR(50) NULL, UNIQUE KEY (email));
INSERT INTO u VALUES (1, NULL), (2, NULL), (3, 'a@x');   -- OK
INSERT INTO u VALUES (4, 'a@x');                          -- ERROR 1062: Duplicate entry 'a@x'
DROP TABLE u;
```

```js
// mg
db.u.createIndex({ email: 1 }, { unique: true });
db.u.insertOne({ _id: 1 });
db.u.insertOne({ _id: 2 });   // E11000 duplicate key error: the missing email is indexed as null
db.u.drop();
db.up.createIndex({ k: 1 }, { unique: true });
db.up.updateOne({ k: 1 }, { $inc: { n: 1 } }, { upsert: true });
db.up.updateOne({ k: 1 }, { $inc: { n: 1 } }, { upsert: true });
db.up.find();                  // one document, n: 2
db.up.drop();
```
