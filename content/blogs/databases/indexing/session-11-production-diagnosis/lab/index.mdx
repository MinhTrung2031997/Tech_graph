### Return to the starting point

The Session 9 lab (and Session 10) left indexes and a foreign key that would hide the problems this case study finds. The scripts below keep the primary keys plus `orders (customer_id)` and `(status, created_at)`, drop the foreign key (MySQL cannot keep it without an index on `order_id`), and delete rows added by earlier runs of this lab. Run them again whenever you want to restart the case study.

```sql
-- pg-s11-reset.sql: docker exec -i idx-pg psql -U postgres -q < pg-s11-reset.sql
ALTER TABLE order_items DROP CONSTRAINT IF EXISTS order_items_order_fk;
DO $$ DECLARE r record; BEGIN
  FOR r IN SELECT i.indexrelid::regclass::text AS idx
           FROM pg_index i JOIN pg_class t ON t.oid = i.indrelid
           WHERE t.relname IN ('orders', 'customers', 'order_items')
             AND NOT EXISTS (SELECT 1 FROM pg_constraint c WHERE c.conindid = i.indexrelid)   -- keep primary keys
             AND i.indexrelid::regclass::text NOT IN ('orders_customer_idx', 'orders_status_created_idx')
  LOOP EXECUTE format('DROP INDEX %s', r.idx); END LOOP;
END $$;
CREATE INDEX IF NOT EXISTS orders_customer_idx ON orders (customer_id);
CREATE INDEX IF NOT EXISTS orders_status_created_idx ON orders (status, created_at);
-- Remove orders and items inserted by earlier runs of the checkout script.
DELETE FROM order_items WHERE id > 2000000;
DELETE FROM orders WHERE id > 1000000;
DROP SEQUENCE IF EXISTS wl_orders, wl_items;
CREATE SEQUENCE wl_orders START 1000001;
CREATE SEQUENCE wl_items START 2000001;
ALTER DATABASE postgres RESET random_page_cost;
VACUUM ANALYZE orders;
VACUUM ANALYZE order_items;
VACUUM ANALYZE customers;
```

### The workload

PostgreSQL: save the 8 pgbench scripts in a folder `wl/`, then `docker cp wl idx-pg:/wl`.

```sql
-- wl/q1_login.sql: find a customer by email
\set c random(1, 100000)
SELECT id, tier, country FROM customers WHERE email = 'c' || :c || '@example.com';

-- wl/q2_history.sql: a customer's newest orders (first page of a keyset list)
\set c random(1, 100000)
SELECT id, created_at, status, amount FROM orders
WHERE customer_id = :c AND created_at < '2026-01-01' ORDER BY created_at DESC LIMIT 20;

-- wl/q3_detail.sql: one order with its items
\set id random(1, 1000000)
SELECT o.id, o.status, o.amount, i.sku, i.qty, i.price
FROM orders o JOIN order_items i ON i.order_id = o.id WHERE o.id = :id;

-- wl/q4_admin.sql: newest pending orders of one day
\set d random(0, 360)
SELECT id, customer_id, amount, created_at FROM orders
WHERE status = 'pending'
  AND created_at >= timestamptz '2025-01-01' + :d * interval '1 day'
  AND created_at <  timestamptz '2025-01-02' + :d * interval '1 day'
ORDER BY created_at DESC LIMIT 50;

-- wl/q5_daily.sql: daily revenue report for one week
\set w random(0, 50)
SELECT date_trunc('day', created_at) AS day, count(*), sum(amount) FROM orders
WHERE created_at >= timestamptz '2025-01-01' + :w * interval '7 days'
  AND created_at <  timestamptz '2025-01-08' + :w * interval '7 days'
GROUP BY 1 ORDER BY 1;

-- wl/q6_topsku.sql: top 10 SKUs of one week
\set w random(0, 50)
SELECT i.sku, sum(i.qty) AS units FROM orders o JOIN order_items i ON i.order_id = o.id
WHERE o.created_at >= timestamptz '2025-01-01' + :w * interval '7 days'
  AND o.created_at <  timestamptz '2025-01-08' + :w * interval '7 days'
GROUP BY i.sku ORDER BY units DESC LIMIT 10;

-- wl/q7_latest.sql: latest order of each gold customer in one country
\set k random(0, 4)
SELECT c.id, o.id AS order_id, o.created_at, o.amount
FROM customers c
CROSS JOIN LATERAL (SELECT id, created_at, amount FROM orders
                    WHERE customer_id = c.id ORDER BY created_at DESC LIMIT 1) o
WHERE c.tier = 'gold' AND c.country = (ARRAY['VN','TH','SG','MY','PH'])[1 + :k];

-- wl/q8_checkout.sql: insert one order and its two items
\set c random(1, 100000)
BEGIN;
INSERT INTO orders VALUES (nextval('wl_orders'), :c, 'pending', 'web', 'SKU-000042', 59.90, now());
INSERT INTO order_items VALUES (nextval('wl_items'), currval('wl_orders'), 'SKU-000042', 1, 29.95),
                               (nextval('wl_items'), currval('wl_orders'), 'SKU-000777', 1, 29.95);
COMMIT;
```

```bash
#!/bin/bash
# run-pg.sh: reset this database's statistics, run 60 s with 4 clients, print the ranking.
P() { docker exec idx-pg psql -U postgres "$@"; }
P -Atc "SELECT pg_stat_statements_reset(0, (SELECT oid FROM pg_database WHERE datname = current_database()), 0)" >/dev/null
docker exec idx-pg sh -c 'cd /wl && pgbench -U postgres -n -c 4 -j 2 -T 60 \
  -f q1_login.sql@30 -f q2_history.sql@25 -f q3_detail.sql@20 -f q4_admin.sql@4 \
  -f q5_daily.sql@1 -f q6_topsku.sql@1 -f q7_latest.sql@2 -f q8_checkout.sql@17 postgres' | grep -E "processed|^tps"
P -c "SELECT left(regexp_replace(query, '\s+', ' ', 'g'), 45) AS query, calls,
       round(total_exec_time) AS total_ms, round(mean_exec_time::numeric, 2) AS mean_ms,
       round(100 * total_exec_time / sum(total_exec_time) OVER ()) AS pct,
       (shared_blks_hit + shared_blks_read) / calls AS blks_per_call
FROM pg_stat_statements
WHERE dbid = (SELECT oid FROM pg_database WHERE datname = current_database())
  AND query NOT LIKE '%pg_stat_statements%'
ORDER BY total_exec_time DESC LIMIT 10"
```

MongoDB: a seeded loop with the same mix (set `N` and the first new order id `OID` when you run it).

```js
// mongo-wl.js: docker cp mongo-wl.js idx-mongo:/ && docker exec idx-mongo mongosh --quiet lab --eval 'const N = 300; let OID = 1000001;' --file /mongo-wl.js
const rnd = (() => { let a = 12345; return () => { a |= 0; a = a + 0x6D2B79F5 | 0; let t = Math.imul(a ^ a >>> 15, 1 | a);
  t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; }; })();   // seeded PRNG
const day = d => new Date(Date.UTC(2025, 0, 1) + d * 86400000);
const countries = ['VN', 'TH', 'SG', 'MY', 'PH'];
let timeouts = 0;
for (let k = 0; k < N; k++) {
  const r = rnd() * 100, c = 1 + Math.floor(rnd() * 100000), id = 1 + Math.floor(rnd() * 1000000);
  const d = Math.floor(rnd() * 361), w = Math.floor(rnd() * 51);
  try {
    if (r < 30) db.customers.findOne({ email: `c${c}@example.com` }, { tier: 1, country: 1 });
    else if (r < 55) db.orders.find({ customer_id: c, created_at: { $lt: new Date('2026-01-01') } },
      { created_at: 1, status: 1, amount: 1 }).sort({ created_at: -1 }).limit(20).toArray();
    else if (r < 75) db.orders.aggregate([{ $match: { _id: id } },
      { $lookup: { from: 'order_items', localField: '_id', foreignField: 'order_id', as: 'items' } }]).toArray();
    else if (r < 79) db.orders.find({ status: 'pending', created_at: { $gte: day(d), $lt: day(d + 1) } },
      { customer_id: 1, amount: 1, created_at: 1 }).sort({ created_at: -1 }).limit(50).toArray();
    else if (r < 80) db.orders.aggregate([{ $match: { created_at: { $gte: day(w * 7), $lt: day(w * 7 + 7) } } },
      { $group: { _id: { $dateTrunc: { date: '$created_at', unit: 'day' } }, n: { $sum: 1 }, revenue: { $sum: '$amount' } } },
      { $sort: { _id: 1 } }]).toArray();
    else if (r < 81) db.orders.aggregate([{ $match: { created_at: { $gte: day(w * 7), $lt: day(w * 7 + 7) } } },
      { $lookup: { from: 'order_items', localField: '_id', foreignField: 'order_id', as: 'items' } },
      { $unwind: '$items' }, { $group: { _id: '$items.sku', units: { $sum: '$items.qty' } } },
      { $sort: { units: -1 } }, { $limit: 10 }], { maxTimeMS: 10000 }).toArray();
    else if (r < 83) db.customers.aggregate([{ $match: { tier: 'gold', country: countries[Math.floor(rnd() * 5)] } },
      { $lookup: { from: 'orders', let: { c: '$_id' }, as: 'latest', pipeline: [
        { $match: { $expr: { $eq: ['$customer_id', '$$c'] } } }, { $sort: { created_at: -1 } }, { $limit: 1 }] } }]).toArray();
    else {
      db.orders.insertOne({ _id: OID, customer_id: c, status: 'pending', channel: 'web', sku: 'SKU-000042', amount: 59.9, created_at: new Date() });
      db.order_items.insertMany([{ _id: 2 * OID - 1, order_id: OID, sku: 'SKU-000042', qty: 1, price: 29.95 },
                                 { _id: 2 * OID, order_id: OID, sku: 'SKU-000777', qty: 1, price: 29.95 }]);
      OID++;
    }
  } catch (e) { if (e.codeName === 'MaxTimeMSExpired') timeouts++; else throw e; }
}
print('timeouts', timeouts, 'next OID', OID);
```
